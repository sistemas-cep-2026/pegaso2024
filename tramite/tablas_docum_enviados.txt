/************* para actualizar tram_documentos_recibidos_seguimiento cc_docrecibidos **********/
UPDATE trs
SET trs.cc_docrecibidos = tdr.cc_docrecibidos
FROM dbo.tram_documentos_recibidos_seguimiento trs

ON trs.c_expediente = tdr.c_expediente;

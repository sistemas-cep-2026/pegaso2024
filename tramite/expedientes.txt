-- 1.1 SEGUIMIENTO: es el más urgente. Sin él, cada EXISTS, cada MAX()
--     y listaIDExp recorren toda la tabla de seguimiento.
CREATE NONCLUSTERED INDEX IX_seg_docrecibidos_area
ON dbo.seguimiento_documentos_recibid1 (cc_docrecibidos, c_area_adm);
GO

-- 1.2 Para listarDerivacion cuando el filtro fuerte es el área
CREATE NONCLUSTERED INDEX IX_seg_area
ON dbo.seguimiento_documentos_recibid1 (c_area_adm, cc_docrecibidos);
GO

-- 1.3 DOCUMENTOS: filtro por local + orden/rango por expediente
--     (listar, listarParaImprimirRango, listarDerivacion, listarTramitesColegiacion)
CREATE NONCLUSTERED INDEX IX_docrec_local_expediente
ON dbo.documentos_recibidos1 (c_local, c_expediente)
INCLUDE (flag, fecha_ingreso, c_local_origen, c_persona, c_persona_pf);
GO

-- 1.4 listar sin local: rango/orden por expediente
CREATE NONCLUSTERED INDEX IX_docrec_expediente
ON dbo.documentos_recibidos1 (c_expediente);
GO

-- 1.5 Dashboards y filtros por fecha (requiere el cambio de YEAR() en PHP)
CREATE NONCLUSTERED INDEX IX_docrec_fecha_local
ON dbo.documentos_recibidos1 (fecha_ingreso, c_local)
INCLUDE (flag);
GO

-- 1.6 Pendientes (listarDerivacion sin expediente: flag = 'P')
CREATE NONCLUSTERED INDEX IX_docrec_pendientes
ON dbo.documentos_recibidos1 (c_local, cc_docrecibidos)
WHERE flag = 'P';
GO

ON dbo.documentos_recibidos1 (c_local, tokenumero);

-- 2.1 nchar(150) ocupa SIEMPRE 300 bytes por fila, haya archivo o no.
ALTER TABLE dbo.seguimiento_documentos_recibid1
ALTER COLUMN file_adjunto NVARCHAR(150) NULL;
GO
UPDATE dbo.seguimiento_documentos_recibid1
SET file_adjunto = RTRIM(file_adjunto)
WHERE file_adjunto IS NOT NULL;
GO

-- 2.2 TEXT (obsoleto) -> VARCHAR(MAX)
ALTER TABLE dbo.seguimiento_documentos_recibid1 ALTER COLUMN observacion    VARCHAR(MAX) NULL;
ALTER TABLE dbo.seguimiento_documentos_recibid1 ALTER COLUMN accion_a_tomar VARCHAR(MAX) NULL;
GO
UPDATE dbo.seguimiento_documentos_recibid1
SET observacion = observacion, accion_a_tomar = accion_a_tomar;  -- mueve los datos dentro de la fila
GO

-- 2.3 decimal(18,0) = 9 bytes; INT = 4 bytes (si 'linea' es un correlativo)
ALTER TABLE dbo.seguimiento_documentos_recibid1 ALTER COLUMN linea INT NULL;
GO

-- 2.4 Un seguimiento sin documento no tiene sentido: NOT NULL + FK
--     (verifica antes: SELECT COUNT(*) FROM seguimiento_documentos_recibid1 WHERE cc_docrecibidos IS NULL)
--     Si hay índices sobre la columna, el ALTER COLUMN exige crearlos DESPUÉS: ejecuta 2.4 antes del bloque 1.
ALTER TABLE dbo.seguimiento_documentos_recibid1 ALTER COLUMN cc_docrecibidos INT NOT NULL;
GO
ALTER TABLE dbo.seguimiento_documentos_recibid1
ADD CONSTRAINT FK_seg_documentos_recibidos
FOREIGN KEY (cc_docrecibidos) REFERENCES dbo.documentos_recibidos1 (cc_docrecibidos);
GO

UPDATE s
SET s.cc_docrecibidos = d.cc_docrecibidos
FROM seguimiento_documentos_recibid s
INNER JOIN documentos_recibidos d
    ON RTRIM(LTRIM(s.c_expediente)) = RTRIM(LTRIM(d.c_expediente))
    AND s.c_local = d.c_local
WHERE s.cc_docrecibidos IS NULL
  AND d.cc_docrecibidos IS NOT NULL;

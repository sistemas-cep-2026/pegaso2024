        echo $row["valor"];

}

public function ArrayLista($sql)
{
    $data = array();
    $rs   = $this->sql_query($sql);
    if (!$rs) {
        die("Error Query: ".$sql."=>".print_r(sqlsrv_errors(), true));
        exit();
    }
    while ($row = $this->sql_fetch_assoc($rs)) {
        $data[] = $row;
    }
    return $data;
}

public function ArrayListaPagina($select, $from, $where, $group, $order,
 $pagina, $por_pagina)
{
    $data = array();
    $sql  = "select * from(
    ".$select."
    ,RowId=ROW_NUMBER() OVER( ".$order.")
    ".$from."
    ".$where."
    ".$group.") tabla where RowId between "
    ."(".$por_pagina." * (".$pagina."-1)+1) and (".$pagina." * ".$por_pagina.")";

    $rs = $this->sql_query($sql);
    if ($rs) {
        while ($row = $this->sql_fetch_assoc($rs)) {
            $data[] = $row;
        }
    }
    return $data;
}

public function sql_fecha($fecha)
{
    $fechaAct = "null";
    if (strlen($fecha) > 0) {
        if ($fecha != "0000-00-00") {
            $fechaFormat = explode("/", $fecha);
            $fechaAct    = "'".$fechaFormat[2]."-".$fechaFormat[1]."-".$fechaFormat[0]."'";
        }
    }
    return $fechaAct;
}

public function lista_campo($tabla)
{
    $db = "BETA202501";
    $data = array();
    $sql  = "SELECT 
    t.TABLE_NAME AS tabla, 
    c.COLUMN_NAME AS campo, 
    c.DATA_TYPE AS tipo_dato, 
    c.CHARACTER_MAXIMUM_LENGTH AS longitud, 
    c.IS_NULLABLE AS permite_nulos
    FROM $db.INFORMATION_SCHEMA.COLUMNS c
    JOIN $db.INFORMATION_SCHEMA.TABLES t 
    ON c.TABLE_NAME = t.TABLE_NAME
    WHERE t.TABLE_TYPE = 'BASE TABLE'
    AND t.TABLE_NAME = '$tabla'
    ORDER BY t.TABLE_NAME, c.ORDINAL_POSITION;";
    $rs = $this->sql_query($sql);
    if ($rs) {
        while ($row = $this->sql_fetch_assoc($rs)) {
            $data[] = $row;
        }
    }
    return $data;
}

//********************************************************* MIGRACION GUNTAS LAS TRES TABLAS*/

SET NOCOUNT ON;
SET XACT_ABORT ON;

DECLARE @tablas TABLE (
    id INT PRIMARY KEY,
    origen SYSNAME,
    destino SYSNAME
);

INSERT INTO @tablas VALUES
(1, 'documentos_de_venta', 'documentos_de_venta2026'),
(2, 'items_planilla_de_cobranza', 'items_planilla_de_cobranza2026'),
(3, 'items_documentos_de_venta', 'items_documentos_de_venta2026');

DECLARE @id INT;
DECLARE @origen SYSNAME;
DECLARE @destino SYSNAME;
DECLARE @columnas NVARCHAR(MAX);
DECLARE @columnas_select NVARCHAR(MAX);
DECLARE @sql NVARCHAR(MAX);
DECLARE @total INT;
DECLARE @error NVARCHAR(4000);

DECLARE @resultados TABLE (
    tabla_destino SYSNAME,
    registros_migrados INT
);

-- ============================================
-- 1. VERIFICAR QUE EXISTEN TODAS LAS TABLAS
-- ============================================
IF EXISTS (
    SELECT 1
    FROM @tablas
    WHERE OBJECT_ID('dbo.' + origen, 'U') IS NULL
       OR OBJECT_ID('dbo.' + destino, 'U') IS NULL
)
BEGIN
    RAISERROR('Falta una tabla origen o destino.',16,1);
    RETURN;
END;

-- ============================================
-- 2. VERIFICAR QUE DESTINOS ESTEN VACIOS
-- ============================================
SET @id = 1;

WHILE @id <= 3
BEGIN
    SELECT
        @origen = origen,
        @destino = destino
    FROM @tablas
    WHERE id = @id;

    SET @sql = N'
        IF EXISTS (
            SELECT 1 FROM dbo.' + QUOTENAME(@destino) + N'
        )
        BEGIN
            RAISERROR(
                ''La tabla destino no esta vacia.'',
                16, 1
            );
        END;
    ';

    EXEC sp_executesql @sql;

    IF @@ERROR <> 0
        RETURN;

    -- Validar igualdad de columnas y propiedades
    IF EXISTS (
        SELECT
            c.name,
            c.system_type_id,
            c.max_length,
            c.precision,
            c.scale,
            c.is_nullable,
            c.is_identity
        FROM sys.columns c
        WHERE c.object_id = OBJECT_ID('dbo.' + @origen)

        EXCEPT

        SELECT
            c.name,
            c.system_type_id,
            c.max_length,
            c.precision,
            c.scale,
            c.is_nullable,
            c.is_identity
        FROM sys.columns c
        WHERE c.object_id = OBJECT_ID('dbo.' + @destino)
    )
    OR EXISTS (
        SELECT name
        FROM sys.columns
        WHERE object_id = OBJECT_ID('dbo.' + @destino)

        EXCEPT

        SELECT name
        FROM sys.columns
        WHERE object_id = OBJECT_ID('dbo.' + @origen)
    )
    BEGIN
        RAISERROR(
            'Las estructuras origen y destino no coinciden.',
            16, 1
        );
        RETURN;
    END;

    SET @id = @id + 1;
END;

-- ============================================
-- 3. INICIAR MIGRACION COMPLETA
-- ============================================
BEGIN TRY

    BEGIN TRANSACTION;

    SET @id = 1;

    WHILE @id <= 3
    BEGIN

        SELECT
            @origen = origen,
            @destino = destino
        FROM @tablas
        WHERE id = @id;

        -- Columnas destino, en su propio orden
        SELECT @columnas = STUFF((
            SELECT ',' + QUOTENAME(c.name)
            FROM sys.columns c
            WHERE c.object_id = OBJECT_ID('dbo.' + @destino)
              AND c.is_computed = 0
            ORDER BY c.column_id
            FOR XML PATH(''), TYPE
        ).value('.', 'NVARCHAR(MAX)'), 1, 1, '');

        -- Columnas origen, por nombre
        SELECT @columnas_select = STUFF((
            SELECT ',o.' + QUOTENAME(c.name)
            FROM sys.columns c
            WHERE c.object_id = OBJECT_ID('dbo.' + @destino)
              AND c.is_computed = 0
            ORDER BY c.column_id
            FOR XML PATH(''), TYPE
        ).value('.', 'NVARCHAR(MAX)'), 1, 1, '');

        -- Migrar todos los registros sin filtro de fecha
        SET @sql = N'
            SET IDENTITY_INSERT dbo.' +
                QUOTENAME(@destino) + N' ON;

            INSERT INTO dbo.' + QUOTENAME(@destino) + N' (
                ' + @columnas + N'
            )
            SELECT ' + @columnas_select + N'
            FROM dbo.' + QUOTENAME(@origen) + N' o;

            SET @total = @@ROWCOUNT;

            SET IDENTITY_INSERT dbo.' +
                QUOTENAME(@destino) + N' OFF;
        ';

        EXEC sp_executesql
            @sql,
            N'@total INT OUTPUT',
            @total OUTPUT;

        INSERT INTO @resultados
        VALUES (@destino, @total);

        SET @id = @id + 1;

    END;

    COMMIT TRANSACTION;

    SELECT
        tabla_destino,
        registros_migrados
    FROM @resultados
    ORDER BY
        CASE tabla_destino
            WHEN 'documentos_de_venta2026' THEN 1
            WHEN 'items_planilla_de_cobranza2026' THEN 2
            ELSE 3
        END;

END TRY
BEGIN CATCH

    SET @error = ERROR_MESSAGE();

    IF @@TRANCOUNT > 0
        ROLLBACK TRANSACTION;

    -- Desactivar IDENTITY_INSERT si quedo activo
    IF @destino IS NOT NULL
    BEGIN
        BEGIN TRY
            SET @sql = N'
                SET IDENTITY_INSERT dbo.' +
                QUOTENAME(@destino) + N' OFF;';
            EXEC sp_executesql @sql;
        END TRY
        BEGIN CATCH
        END CATCH;
    END;

    RAISERROR(@error,16,1);

END CATCH;

SET NOCOUNT ON;
SET XACT_ABORT ON;

DECLARE @planillas INT;
DECLARE @detalles INT;
DECLARE @error NVARCHAR(4000);

-- Verificar duplicados en la tabla principal
IF EXISTS (
    SELECT 1
    FROM dbo.documentos_de_venta2026
    GROUP BY cod_documento, num_documento
    HAVING COUNT(*) > 1
)
BEGIN
    RAISERROR(
        'Existen comprobantes duplicados. No se puede actualizar.',
        16, 1
    );
    RETURN;
END;

BEGIN TRY
    BEGIN TRANSACTION;

    -- =========================================
    -- 1. ACTUALIZAR ITEMS PLANILLA DE COBRANZA
    -- =========================================
    UPDATE p
    SET p.cc_ventas = v.cc_ventas
    FROM dbo.items_planilla_de_cobranza2026 p
    INNER JOIN dbo.documentos_de_venta2026 v
        ON p.cod_documento = v.cod_documento
       AND p.num_documento = v.num_documento
    WHERE p.cc_ventas IS NULL
       OR p.cc_ventas <> v.cc_ventas;

    SET @planillas = @@ROWCOUNT;

    -- =========================================
    -- 2. ACTUALIZAR ITEMS DOCUMENTOS DE VENTA
    -- =========================================
    UPDATE i
    SET i.cc_ventas = v.cc_ventas
    FROM dbo.items_documentos_de_venta2026 i
    INNER JOIN dbo.documentos_de_venta2026 v
        ON i.cod_documento = v.cod_documento
       AND i.num_documento = v.num_documento
    WHERE i.cc_ventas IS NULL
       OR i.cc_ventas <> v.cc_ventas;

    SET @detalles = @@ROWCOUNT;

    COMMIT TRANSACTION;

    -- =========================================
    -- 3. MOSTRAR RESULTADOS
    -- =========================================
    SELECT
        @planillas AS planillas_actualizadas,
        @detalles AS detalles_actualizados,
        'ACTUALIZACION COMPLETADA' AS resultado;

END TRY
BEGIN CATCH

    IF @@TRANCOUNT > 0
        ROLLBACK TRANSACTION;

    SET @error = ERROR_MESSAGE();

    RAISERROR(@error,16,1);

END CATCH;
